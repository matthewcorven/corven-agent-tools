"""Read a YouTube API key without echo and verify the existing research metadata."""
import datetime
import getpass
import json
from pathlib import Path
import resource
import ssl
import sys
import urllib.error
import urllib.parse
import urllib.request
import warnings

ROOT = Path('/Users/core/git/matthewcorven/corven-agents/research/indydevdan/2026-11-10')
CHANNEL_ID = 'UC_x36zCEGilGpB1m-V4gmjg'
API_BASE = 'https://www.googleapis.com/youtube/v3/'
RESULT = Path(__file__).with_name('youtube-api-verification-result.json')


class NoRedirects(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def main():
    resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
    warnings.simplefilter('error', getpass.GetPassWarning)
    if not sys.stdin.isatty() or not sys.stdout.isatty():
        print('Stopped: an interactive terminal is required for hidden input.')
        return 1
    print('Enter the YouTube API key below. Input will not be echoed.')
    print('It will be used only in this process for read-only requests to Google,')
    print('then discarded. No environment export, file storage, or secret logging.')
    print('This verifies channel/video metadata; it does not download transcripts.')
    key = None
    try:
        key = getpass.getpass('GOOGLE_API_YOUTUBE_API_KEY (hidden): ').strip()
        if not key:
            print('Cancelled: no key supplied.')
            return 1
        selection = json.loads((ROOT / 'selection.json').read_text())
        video_ids = [item['id'] for item in selection['selected_videos']]
        opener = urllib.request.build_opener(
            NoRedirects(), urllib.request.HTTPSHandler(context=ssl.create_default_context()))

        def api(resource_name, parameters):
            url = API_BASE + resource_name + '?' + urllib.parse.urlencode(parameters)
            request = urllib.request.Request(url, headers={
                'X-Goog-Api-Key': key,
                'Accept': 'application/json',
            })
            with opener.open(request, timeout=30) as response:
                return json.load(response)

        channel_response = api('channels', {'part': 'snippet', 'forHandle': '@indydevdan'})
        channels = channel_response.get('items', [])
        if len(channels) != 1 or channels[0]['id'] != CHANNEL_ID:
            print('Stopped: returned channel did not match the research collection.')
            return 1
        video_response = api('videos', {
            'part': 'snippet,contentDetails,status', 'id': ','.join(video_ids),
        })
        videos = {item['id']: item for item in video_response.get('items', [])}
        if set(videos) != set(video_ids):
            print('Stopped: Google did not return the exact six selected videos.')
            return 1
        records = []
        for video_id in video_ids:
            item = videos[video_id]
            snippet = item['snippet']
            if snippet['channelId'] != CHANNEL_ID:
                print('Stopped: a returned video belongs to another channel.')
                return 1
            records.append({
                'video_id': video_id,
                'title': snippet['title'],
                'url': 'https://www.youtube.com/watch?v=' + video_id,
                'published_at': snippet['publishedAt'],
                'duration_iso8601': item['contentDetails']['duration'],
                'privacy_status': item['status']['privacyStatus'],
            })
        result = {
            'status': 'verified',
            'verified_at_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
            'method': 'YouTube Data API v3; key entered with no echo and used only in process memory via HTTPS request headers.',
            'channel': {'id': CHANNEL_ID, 'title': channels[0]['snippet']['title']},
            'videos': records,
            'scope': 'Read-only verification of the six videos already selected from the channel Videos tab. This is not a fresh upload-order check or caption authorization.',
            'credential_stored': False,
        }
        RESULT.write_text(json.dumps(result, indent=2) + '\n')
        print('\nSuccess: Google verified the channel and all six selected videos.')
        print('Only public metadata was saved to the session verification result.')
        print('The key was not saved or exported; this process is exiting.')
        return 0
    except urllib.error.HTTPError as error:
        print(f'\nGoogle API returned HTTP {error.code}. No credential or response body was logged.')
        return 1
    except (EOFError, KeyboardInterrupt):
        print('\nCancelled. No key was saved.')
        return 1
    except getpass.GetPassWarning:
        print('\nStopped: hidden terminal input is unavailable. Nothing was read.')
        return 1
    except Exception as error:
        print(f'\nStopped ({type(error).__name__}); details suppressed to protect the key.')
        return 1
    finally:
        key = None


if __name__ == '__main__':
    raise SystemExit(main())
